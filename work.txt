-- 1. ADD NEW COLUMNS TO THE SERVICES TABLE
ALTER TABLE `services` 
ADD COLUMN IF NOT EXISTS `show_website` TINYINT(1) NOT NULL DEFAULT 1 AFTER `status`,
ADD COLUMN IF NOT EXISTS `show_landing` TINYINT(1) NOT NULL DEFAULT 1 AFTER `show_website`,
ADD COLUMN IF NOT EXISTS `landing_title` VARCHAR(255) NULL AFTER `show_landing`,
ADD COLUMN IF NOT EXISTS `landing_short_description` TEXT NULL AFTER `landing_title`,
ADD COLUMN IF NOT EXISTS `landing_description` TEXT NULL AFTER `landing_short_description`,
ADD COLUMN IF NOT EXISTS `landing_bullet_points` TEXT NULL AFTER `landing_description`;

-- 2. COPY EXISTING WEBSITE DATA INTO LANDING PAGE FIELDS FOR ALL EXISTING SERVICES
UPDATE `services` 
SET 
  `show_website` = 1,
  `show_landing` = 1,
  `landing_title` = `title`,
  `landing_short_description` = `short_description`,
  `landing_description` = `long_description`
WHERE `landing_title` IS NULL OR `landing_title` = '';

-- 3. ADD TO_NAME, DESIGNATION, LOCATION TO LETTERPADS TABLE
ALTER TABLE `letterpads` 
ADD COLUMN IF NOT EXISTS `to_name` VARCHAR(255) NULL AFTER `title`,
ADD COLUMN IF NOT EXISTS `designation` VARCHAR(255) NULL AFTER `to_name`,
ADD COLUMN IF NOT EXISTS `location` VARCHAR(255) NULL AFTER `designation`;

-- 4. ADD TEAM_MEMBER_ID, REGARDS_NAME, REGARDS_ROLE FOR LETTERPAD SIGN-OFF
ALTER TABLE `letterpads`
ADD COLUMN IF NOT EXISTS `team_member_id` INT(11) NULL AFTER `location`,
ADD COLUMN IF NOT EXISTS `regards_name` VARCHAR(255) NULL AFTER `team_member_id`,
ADD COLUMN IF NOT EXISTS `regards_role` VARCHAR(255) NULL AFTER `regards_name`;
